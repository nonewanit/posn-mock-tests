\question ผลลัพธ์ของโปรแกรมนี้ตรงกับข้อใด
\begin{verbatim}
numbers = [2, 3, 4, 5, 6, 7]
n = len(numbers)
i = 0
count = 0
while i < n:
    j = 2
    ch = 1
    while j < numbers[i]:
        if numbers[i] % j == 0:
            ch = 0
        j = j + 1
    if ch == True:
        count = count + 1
    i = i + 1
print(count)
\end{verbatim}
\choiceshorizontal{0}{2}{3}{4}
% answer: 4
