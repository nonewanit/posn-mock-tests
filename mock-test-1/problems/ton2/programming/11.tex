\question ให้โปรแกรม A, B, C และ D ต่อไปนี้ โปรแกรมใดบ้างที่คำนวณและแสดงผลค่าของ $\binom{10}{5}$ ได้ถูกต้อง

\begin{center}
\begin{minipage}[t]{0.45\textwidth}
\textbf{โปรแกรม A}
\begin{verbatim}
n = 10
k = 5
c = 1
i = 1
while i <= k:
    c = c * (n - i + 1) // i
    i = i + 1
print(c)
\end{verbatim}
\end{minipage}
\hfill

\textbf{โปรแกรม B}

i = 0
while i < k:
    c = c * (n - i) // (i + 1)
    i = i + 1
print(c)
\end{verbatim}
\end{minipage}
