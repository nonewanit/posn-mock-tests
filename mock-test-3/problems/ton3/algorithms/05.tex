\question จากโค้ดหา ถ้ารับข้อมูลนำเข้าเป็น aladwdadj จะได้ข้อมูลส่งออกเป็นอะไร
\begin{verbatim}
s = input().strip()
n = len(s)

dp = [[0] * n for i in range(n)]
mx = 1
st = 0

for i in range(n):
    dp[i][i] = 1

for i in range(1, n):
    if s[i] == s[i - 1]:
        dp[i - 1][i] = 1
        mx = 2
        st = i - 1

for length in range(3, n + 1):
    for i in range(n - length + 1):
        j = i + length - 1
        if s[i] == s[j] and dp[i + 1][j - 1]:
            dp[i][j] = 1
            st = i
            mx = length

print(s[st:st + mx])
\end{verbatim}
