\question เมื่อขั้นตอนวิธีต่อไปนี้ทำงานเสร็จ ค่าที่ส่งออกตรงกับข้อใด
\begin{pseudocode}
    \begin{enumerate}
        \item กำหนดให้ a มีค่าเท่ากับ 48 และ b มีค่าเท่ากับ 18
        \item ทำซ้ำในขณะที่ a ไม่เท่ากับ b:
        \begin{enumerate}
            \item ถ้า a มากกว่า b:
            \begin{enumerate}
                \item ให้ a มีค่าเท่ากับ a - b
            \end{enumerate}
            \item มิฉะนั้น:
            \begin{enumerate}
                \item ให้ b มีค่าเท่ากับ b - a
            \end{enumerate}
        \end{enumerate}
        \item ส่งออกค่า a
    \end{enumerate}
\end{pseudocode}
\choiceshorizontal{2}{6}{12}{18}
% answer: 2
