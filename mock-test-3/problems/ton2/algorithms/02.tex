\question ข้อใดให้ผลลัพธ์ตรงกับขั้นตอนวิธีต่อไปนี้
\begin{pseudocode}
\begin{enumerate}
    \item ให้ a มีค่าเท่ากับ 6
    \item ถ้า a หารด้วย 3 เท่ากับ 2 
        \begin{enumerate}
            \item ให้ a มีค่าเท่ากับ 5
        \end{enumerate}
    \item มิฉะนั้นถ้า a หารด้วย 5 มีค่ามากกว่าหรือเท่ากับ 1
        \begin{enumerate}
            \item ให้ a มีค่าเท่ากับ 6
        \end{enumerate}
    \item มิฉะนั้น
        \begin{enumerate}
            \item ให้ a มีค่าเท่ากับ 1
        \end{enumerate}
    \item พิมพ์ค่า a
\end{enumerate}
\end{pseudocode}

\choiceshorizontal{1}{5}{6}{7}
