\begin{verbatim}
n = int(input());
m = int(input());
if(n%m):
    print('A');
elif(n+m>10):
    print('B');
elif(m\*n >= n\*\*5 and n//10 <= 0):
    print('C');
elif(n>m):
    print('D');
else:
    print('E');
\end{verbatim}
\choiceshorizontal{A}{B}{C}{D}
