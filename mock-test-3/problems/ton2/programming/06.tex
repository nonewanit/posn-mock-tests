\question จากโค้ดนี้มีการแสดงผลตัวอักษร \texttt{*} ทั้งหมดกี่ตัว
\begin{verbatim}
i = 0
while i < 5:
    i = i + 1
    j = 0
    while j < 5:
        j = j + 1
        if (i + j) % 2:
            print('*')
        else:
            print(' ')
\end{verbatim}
\choiceshorizontal{9}{10}{11}{12}
% answer: 4
