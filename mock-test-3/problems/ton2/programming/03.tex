\question \begin{tabular}[t]{@{}l@{}}โปรเเกรมต่อไปนี้ จะเเสดงผลตรงกับข้อใด
\\เมื่อ \% ให้ผลลัพธ์เป็นเศษเหลือของการหาร เช่น 3\%2 ได้ผลลัพธ์เป็น 1
\end{tabular}
\begin{verbatim}
a=0
b=1
i=1
while i<=3:
    a=a+b*(i%2)
    b=b+a*(1-i%2)
    i=i+1
print(a)
\end{verbatim}

\choiceshorizontal{2}{3}{5}{8}
% answer: 2
