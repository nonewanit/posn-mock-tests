\begin{verbatim}
def decode(s):
    n = len(s)
    i = 0
    result = ""
    while i < n:
        char_to_repeat = s[i]
        i += 1
        
        count_str = ""
        while i < n and s[i].isdigit():
            count_str += s[i]
            i += 1
        
        count = int(count_str)
        result += char_to_repeat * count
        
    return result

s = input()
print(decode(s))
