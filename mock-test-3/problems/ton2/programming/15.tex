\question จากโค้ดนี้ ถ้าใช้ s ข้อใดจะทำให้ไม่มีข้อความนำออกเป็น "Error" (กำหนดให้ x.isdigit() จะส่งค่ากลับเป็น 1 ถ้า x เป็นตัวเลข และ 0 ถ้า x ไม่ใช่ตัวเลข)
\begin{verbatim}
s = input()
n = len(s)
i = 0
result = ""
error = False
while i < n:
    char_to_repeat = s[i]
    count_str = 0
    i = i + 1
    if s[i].isdigit()==0:
        error = True
        break
    while i < n and s[i].isdigit():
        count_str = count_str*10 + int(s[i])
        i = i + 1
    
    x = 0
    while(x<count_str):
        result += char_to_repeat
        x = x + 1
if error:
    print("Error")
else :
    print(result)
\end{verbatim}
