\question จากโค้ดนี้ ถ้าใช้ s ข้อใดจะทำให้ไม่เกิด error (กำหนดให้ x.isdigit() จะส่งค่ากลับเป็น 1 ถ้า x เป็นตัวเลข และ 0 ถ้า x ไม่ใช่ตัวเลข)
\begin{verbatim}
n = len(s)
i = 0
while i < n:
    char_to_repeat = s[i]
    count_str = 0
    i += 1
    while i < n and s[i].isdigit():
        count_str = count_str*10 + int(s[i])
        i += 1
    x = 0
    while(x<count_str):
        print(char_to_repeat,end='')
        x+=1
\end{verbatim}
