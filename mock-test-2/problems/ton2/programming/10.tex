\question โปรแกรมต่อไปนี้ แสดงผลลัพธ์เป็น * จำนวนกี่ครั้ง
\begin{verbatim}
n = 5
m = 4
i = 0
while i < n:
    j = 0
    while j < m:
        if (i+j) % 2 == 0:
            print("*")
        else:
            print("-")
        j = j + 1
    i = i + 1
\end{verbatim}
